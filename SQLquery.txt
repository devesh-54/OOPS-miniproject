-- ----------------------------------------------------------
-- DATABASE : Patient Billing and Pharmacy Management System
-- ----------------------------------------------------------

-- 1️⃣ Create Database and Use It
CREATE DATABASE IF NOT EXISTS hospitaldb;
USE hospitaldb;

-- ----------------------------------------------------------
-- 2️⃣ Create Table: patients
-- ----------------------------------------------------------
CREATE TABLE IF NOT EXISTS patients (
    patient_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    age INT,
    gender VARCHAR(10),
    diagnosis VARCHAR(200),
    contact VARCHAR(15)
);

-- ----------------------------------------------------------
-- 3️⃣ Create Table: medicines
-- ----------------------------------------------------------
CREATE TABLE IF NOT EXISTS medicines (
    medicine_id INT AUTO_INCREMENT PRIMARY KEY,
    medicine_name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2),
    stock INT DEFAULT 0
);

-- ----------------------------------------------------------
-- 4️⃣ Create Table: bills
-- ----------------------------------------------------------
CREATE TABLE IF NOT EXISTS bills (
    bill_id INT AUTO_INCREMENT PRIMARY KEY,
    patient_id INT,
    total_amount DECIMAL(10,2),
    bill_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (patient_id) REFERENCES patients(patient_id)
);

-- ----------------------------------------------------------
-- 🔟 Display Bill Details with Patient Names
-- ----------------------------------------------------------
SELECT b.bill_id, p.name AS patient_name, b.total_amount, b.bill_date
FROM bills b
JOIN patients p ON b.patient_id = p.patient_id;

-- ----------------------------------------------------------
-- 1️⃣1️⃣ Show Total Revenue
-- ----------------------------------------------------------
SELECT SUM(total_amount) AS total_revenue FROM bills;

-- 1️⃣2️⃣ Update Medicine Stock (Example)
-- ----------------------------------------------------------
UPDATE medicines SET stock = stock - 2 WHERE medicine_id = 1;

-- ----------------------------------------------------------
-- 1️⃣3️⃣ Update Patient Contact Info (Example)
-- ----------------------------------------------------------
UPDATE patients SET contact = '9876500000' WHERE patient_id = 1;

-- ----------------------------------------------------------
-- 1️⃣4️⃣ Delete a Bill Record (Example)
-- ----------------------------------------------------------
DELETE FROM bills WHERE bill_id = 3;

-- ----------------------------------------------------------
-- 1️⃣5️⃣ Create a View for Patient Billing Summary
-- ----------------------------------------------------------
CREATE OR REPLACE VIEW patient_billing_summary AS
SELECT p.name, p.diagnosis, b.total_amount, b.bill_date
FROM patients p
JOIN bills b ON p.patient_id = b.patient_id;

-- ----------------------------------------------------------
-- 1️⃣6️⃣ Display Summary View
-- ----------------------------------------------------------
SELECT * FROM patient_billing_summary;
